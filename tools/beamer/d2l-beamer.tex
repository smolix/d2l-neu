% D2L Beamer theme for slides_to_beamer.py (included with pandoc -H).
% Proportions follow the PowerPoint export (pptx_fit.py) scaled from a
% 10 x 5.625 in slide to Beamer's 16:9 page (160 x 90 mm, factor 0.63):
% body 11.3 pt (18 pt), title 15.1 pt (24 pt), footer 6.3 pt (10 pt).
% The deck defines \dtllogo, \dtlfooter and \dtlcover before \input-ing
% this file, and then \input's sizes.tex (the fitted per-frame sizes).

% ---- per-frame size -----------------------------------------------------
% \dtlsz@<n> (body pt) and \dtlim@<n> (image height cap, pt) come from the
% fit loop; frames without an entry use the base size.
\makeatletter
\def\dtl@base{11.3}
\def\dtl@s{11.3}
\newlength\dtlimgmax
\setlength\dtlimgmax{0.8\textheight}
\def\dtlcodefont{\fontsize{8.8}{10.2}\selectfont}
\def\dtl@sub{\fontsize{10.4}{12.4}}
\setbeamerfont{itemize/enumerate body}{size=}
\setbeamerfont{itemize/enumerate subbody}{size=\dtl@sub}
\setbeamerfont{itemize/enumerate subsubbody}{size=\dtl@sub}
\newcommand\dtlstart[1]{%
  \ifcsname d2lsz@#1\endcsname
    \edef\dtl@s{\csname d2lsz@#1\endcsname}\else\edef\dtl@s{\dtl@base}\fi
  \ifcsname d2lim@#1\endcsname
    \setlength\dtlimgmax{\csname d2lim@#1\endcsname pt}%
  \else\setlength\dtlimgmax{0.65\textheight}\fi
  \fontsize{\dtl@s}{\fpeval{1.2*\dtl@s}}\selectfont
  \edef\dtlcodefont{\noexpand\fontsize{\fpeval{0.78*\dtl@s}}%
    {\fpeval{0.9*\dtl@s}}\noexpand\selectfont}%
  \edef\dtl@sub{\noexpand\fontsize{\fpeval{0.92*\dtl@s}}{\fpeval{1.1*\dtl@s}}}%
  \setlength\parskip{0.45em}%
  \setlength\abovedisplayskip{0.4em}\setlength\belowdisplayskip{0.4em}%
  \setlength\abovedisplayshortskip{0.2em}\setlength\belowdisplayshortskip{0.4em}%
  \zsavepos{d2ls-#1}}
% End marker: the bottom of the frame's content.
\newcommand\dtlend[1]{\par\zsavepos{d2le-#1}}

% ---- kicker ---------------------------------------------------------------

   \color{d2lblue}\bfseries\addfontfeature{LetterSpace=6}\MakeUppercase{#1}}}

% ---- images: fit the column width and the frame's height cap ---------------
% (pandoc defines \pandocbounded only for decks with images)
\ifdefined\pandoc@box\else\newsavebox\pandoc@box\fi
\providecommand*\pandocbounded[1]{}
\renewcommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{\dtlimgmax}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \leavevmode\hfill\scalebox{\@tempa}{\usebox\pandoc@box}\hfill\null}

% ---- frame title: bottom-anchored, shrunk to one line down to 10.1 pt -----
\newsavebox\dtl@tbox
\newdimen\dtl@ts
\newcommand\dtlfittitle[1]{%
  \sbox\dtl@tbox{\fontsize{15.1}{18}\selectfont\bfseries #1}%
  \dtl@ts=15.1pt
  \ifdim\wd\dtl@tbox>0.985\linewidth
    \dtl@ts=\dimexpr 14.85pt*\numexpr\linewidth\relax/\numexpr\wd\dtl@tbox\relax\relax
    \ifdim\dtl@ts<10.1pt \dtl@ts=10.1pt\fi
  \fi
  \begin{minipage}[b]{\linewidth}\raggedright
    \fontsize{\strip@pt\dtl@ts}{\fpeval{1.12*\strip@pt\dtl@ts}}\selectfont
    \bfseries #1\end{minipage}}
\setbeamertemplate{frametitle}{%
  \parskip=0pt
  \vbox to 1.10cm{\vfill\hbox{\dtlfittitle{\insertframetitle}}\vskip 0.16cm}%
  \nointerlineskip
  \hbox{\color{d2lbluelight}\rule{\linewidth}{0.9pt}}%
  \vskip 0.10cm}
